package learning;

import java.io.*;
import java.util.*;
import java.util.concurrent.atomic.*;

public class Dispatcher {
	public static void main(String[] args) {
		List<String> filesPaths = Arrays.asList("InputFiles/aa.txt", "InputFiles/bb.txt", "InputFiles/cc.txt");
		LongAdder wordsCount = new LongAdder();
		LongAdder lettersCount = new LongAdder();
		List<Thread> threads = new ArrayList<>();
		for(String filePath : filesPaths) {
			threads.add(new Thread(new CountWordsAndLetters(new File(filePath), wordsCount, lettersCount)));
		}
		for(Thread thread : threads) {
			thread.start();
		}
		long startTime = System.currentTimeMillis();
		try {
			for(Thread thread : threads) {
				thread.join();
			}
		}catch(InterruptedException e) {
			e.printStackTrace();
		}
		System.out.println(System.currentTimeMillis() - startTime);
		System.out.println(lettersCount.longValue() / wordsCount.doubleValue());
	}
}

class CountWordsAndLetters implements Runnable {
	final private File file;
	final private LongAdder wordsCount;
	final private LongAdder lettersCount;
	
	CountWordsAndLetters(File file, LongAdder wordsCount, LongAdder lettersCount) {
		this.file = file;
		this.wordsCount = wordsCount;
		this.lettersCount = lettersCount;
	}
	
	public void run() {
		try(BufferedReader reader = new BufferedReader(new FileReader(this.file))) {
			reader.lines().parallel().forEach(line -> {
				boolean isPreviousLetter = false;
				for(char character : line.toCharArray()) {
					if(character == ' ') {
						if(isPreviousLetter) {
							this.wordsCount.increment();
							isPreviousLetter = false;
						}
					}else if(character == ',' || character == '.' || character == '!' || character == '?') {
						this.wordsCount.increment();
						isPreviousLetter = false;
					}else {
						this.lettersCount.increment();
						isPreviousLetter = true;
					}
				}
			});
		}catch(IOException e) {
			e.printStackTrace();
		}
	}
}
