package learning;

import java.util.*;

public class Dispatcher {
	public static void main(String[] args) {
		int[] numbers = {1, -1, 2, -2, 0, 8, -3, 3, 4, -4, 5, -5, 8, 8, 0, -2, 4, 4, 2 , 22, 54, 33, 98, 444, 232, 444};
		
		Map<Integer, Integer> sortedMap = new TreeMap<>(Arrays.stream(numbers).parallel()
										  .filter(num -> num % 2 == 0)
										  .collect(HashMap<Integer, Integer>::new,
												  (map, value) -> map.merge(value, 1, Integer::sum),
												  (map1, map2) -> {
													  for(Map.Entry<Integer, Integer> entry: map2.entrySet()) {
														  map1.merge(entry.getKey(), entry.getValue(), Integer::sum);
													  }
												  }));
		System.out.println(sortedMap);
	}
}
