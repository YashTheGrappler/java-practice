package learning;

import java.util.*;
import java.util.stream.*;

public class Dispatcher {
	public static void main(String[] args) {
		String text = "The class String includes methods for examining individual characters of the sequence, "
				+ "for comparing strings, for searching strings, for extracting substrings, "
				+ "and for creating a copy of a string with all characters translated to uppercase or to lowercase. "
				+ "Case mapping is based on the Unicode Standard version specified by the Character class.";

		String capitalized = Arrays.stream(text.split(" "))
				.map(word -> word.isEmpty() ? word : word.substring(0, 1).toUpperCase() + word.substring(1).toLowerCase())
				.collect(Collectors.joining(" "));
		System.out.println(capitalized);
		
		Arrays.stream(text.split("\\."))
		.map(sentence -> sentence.trim().split(" "))
		.forEach(sentence -> System.out.println(Arrays.toString(sentence) + ". " + sentence.length));
		
		List<Integer> collection = Arrays.asList(1, -1, 2, -2, 0, -3, 3, 4, -4, 5, -5);
		List<List<Integer>> numbers = collection.stream()
				.collect(() -> Arrays.asList(new ArrayList<>(), new ArrayList<>()),
						(alli, num) -> {
							if(num > -1) alli.get(0).add(num);
							else alli.get(1).add(num);
							},
						(list1, list2) -> {
							list1.get(0).addAll(list2.get(0));
						    list1.get(1).addAll(list2.get(1));
						});
		System.out.println(numbers);
	}
}
